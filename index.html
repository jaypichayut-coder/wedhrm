สถาปัตยกรรมการเชื่อมต่อ GitHub Pages + Cloudflare Workers

หน้านี้สรุปวิธีนำไฟล์ HTML นี้ไปรันผ่าน GitHub Repository และรัน Backend เบาๆ ผ่าน Cloudflare Workers

1

GitHub Repository / Pages

อัปโหลดไฟล์ index.html นี้ขึ้น GitHub และเปิดใช้งาน GitHub Pages เพื่อให้ทีม HR เข้าใช้งานได้ทุกที่ตลอดเวลา

2

Cloudflare Worker (API / Database)

ทำหน้าที่เป็น REST API Proxy รับส่งข้อมูลสถานะผู้สมัคร (JSON / D1 / KV / Google Sheets Sync) ด้วย latency ต่ำและไม่มีค่าใช้จ่ายเซิร์ฟเวอร์

3

Client Data Sync

ระบบเว็บเชื่อมต่ออัตโนมัติ มีโหมด Standalone ออฟไลน์พร้อมรองรับการนำเข้าไฟล์ CSV เดิมจากแบบฟอร์มสมัครงาน

ตัวอย่างโค้ด Cloudflare Worker (worker.js) 
/**
 * LNC+ Recruitment Cloudflare Worker API
 * รองรับการอ่าน/บันทึกสถานะผู้สมัคร (ผ่าน / รอสัมภาษณ์) และข้อมูลสาขา
 */
export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    
    // ตั้งค่า CORS headers เพื่อให้ GitHub Pages เรียกใช้งานได้
    const corsHeaders = {
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
      'Access-Control-Allow-Headers': 'Content-Type, Authorization',
    };

    if (request.method === 'OPTIONS') {
      return new Response(null, { headers: corsHeaders });
    }

    // Endpoint: GET /api/candidates
    if (url.pathname === '/api/candidates' && request.method === 'GET') {
      // สามารถดึงจาก Cloudflare KV หรือ D1 Database
      // ตัวอย่างการดึงจาก KV: const data = await env.LNC_RECRUIT_KV.get('candidates');
      return new Response(JSON.stringify({ status: 'ok', data: [] }), {
        headers: { ...corsHeaders, 'Content-Type': 'application/json' }
      });
    }

    // Endpoint: POST /api/candidates (บันทึกหรืออัปเดตสถานะช่อง X)
    if (url.pathname === '/api/candidates' && request.method === 'POST') {
      const payload = await request.json();
      // ตัวอย่าง: await env.LNC_RECRUIT_KV.put('candidates', JSON.stringify(payload));
      return new Response(JSON.stringify({ status: 'saved', count: payload.length }), {
        headers: { ...corsHeaders, 'Content-Type': 'application/json' }
      });
    }

    return new Response(JSON.stringify({ message: "LNC+ Recruitment Worker API Active" }), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' }
    });
  }
};
